cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(ConcordS33Diagnostics LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT MSVC)
    message(FATAL_ERROR "S33 diagnostics require the verified Windows clang-cl toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(concord_s33_capabilities Capabilities.cpp)
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(concord_s33_capabilities PRIVATE /W4 /WX /utf-8)
endif()
target_compile_definitions(concord_s33_capabilities PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(concord_s33_capabilities PRIVATE d3d11 dxgi dwmapi psapi)
add_executable(concord_s33_timing_contract TimingContract.cpp)
target_include_directories(concord_s33_timing_contract PRIVATE ../../../src)
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(concord_s33_timing_contract PRIVATE /W4 /WX /utf-8)
endif()
enable_testing()
add_test(NAME s33_timing_contract COMMAND concord_s33_timing_contract)
set_tests_properties(s33_timing_contract PROPERTIES ENVIRONMENT
    "CONCORD_LIVE_BENCHMARK=1;CONCORD_LIVE_BENCHMARK_TICKS=60;CONCORD_LIVE_BENCHMARK_VISIBLE=1;CONCORD_LIVE_BENCHMARK_RAW_SAMPLES=1")
