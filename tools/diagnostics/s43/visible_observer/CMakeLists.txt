cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(ConcordS43VisibleObserver LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT MSVC)
    message(FATAL_ERROR "S43 observer requires the verified Windows clang-cl toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(concord_s43_visible_observer VisibleObserver.cpp)
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(concord_s43_visible_observer PRIVATE /W4 /WX /utf-8)
endif()
target_compile_definitions(concord_s43_visible_observer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(concord_s43_visible_observer PRIVATE d3d11 dxgi dwmapi psapi)
