cmake_minimum_required(VERSION 3.20)
project(ConcordBinkDecoder LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "BinkFrameExtractor must be built as a 32-bit Windows executable")
endif()
add_executable(BinkFrameExtractor BinkFrameExtractor.cpp)
target_compile_features(BinkFrameExtractor PRIVATE cxx_std_17)
if(MSVC)
    set_property(TARGET BinkFrameExtractor PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    target_compile_options(BinkFrameExtractor PRIVATE /utf-8)
endif()
