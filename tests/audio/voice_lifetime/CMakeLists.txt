# Only native verification uses device libraries. The existing production PImpl
# is compiled unchanged with each matching producer/consumer ABI.
include("${CMAKE_SOURCE_DIR}/cmake/CompilerContract.cmake")
if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
if(WIN32 AND NOT CONCORD_PORTABLE_CORE)
    foreach(voice_pcm_profile IN ITEMS runtime contract)
        set(voice_pcm_target concord_voice_pcm_lifetime_${voice_pcm_profile})
        add_executable(${voice_pcm_target} VoicePcmLifetimeContract.cpp)
        target_include_directories(${voice_pcm_target} PRIVATE "${CMAKE_SOURCE_DIR}/src")
        target_compile_features(${voice_pcm_target} PRIVATE cxx_std_20)
        if(voice_pcm_profile STREQUAL "runtime")
            target_compile_options(${voice_pcm_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime,$<TARGET_PROPERTY:concord_runtime,COMPILE_OPTIONS>>")
            target_compile_definitions(${voice_pcm_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime,$<TARGET_PROPERTY:concord_runtime,COMPILE_DEFINITIONS>>")
            concord_runtime_crt_profile(${voice_pcm_target})
            target_link_libraries(${voice_pcm_target} PRIVATE concord_audio_diagnostic_native_leaf)
        else()
            target_compile_options(${voice_pcm_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_OPTIONS>>")
            target_compile_definitions(${voice_pcm_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_DEFINITIONS>>")
            # The unchanged production PImpl has the Runtime ABI in every build.
            concord_runtime_abi_definition(${voice_pcm_target})
            concord_runtime_crt_profile(${voice_pcm_target})
            target_link_libraries(${voice_pcm_target} PRIVATE concord_audio_diagnostic_native_leaf)
        endif()
        add_test(NAME voice_pcm_lifetime_${voice_pcm_profile}_test COMMAND ${voice_pcm_target})
        set_tests_properties(voice_pcm_lifetime_${voice_pcm_profile}_test PROPERTIES
            TIMEOUT 120 RUN_SERIAL TRUE LABELS "windows;voice-pcm;real-native-device;explicit-output")
    endforeach()
    unset(voice_pcm_profile)
    unset(voice_pcm_target)
endif()
