if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_audio_session_profile IN ITEMS runtime contract)
    if(_audio_session_profile STREQUAL "runtime")
        set(_audio_session_owner concord_runtime)
        set(_audio_session_crt MultiThreadedDLL)
    else()
        set(_audio_session_owner concord_radar_cell_values_value_contract)
        set(_audio_session_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_audio_session_test concord_world_audio_backend_session_${_audio_session_profile}_contract)
    add_executable(${_audio_session_test} "${PROJECT_SOURCE_DIR}/tests/audio/world_backend_session/RuntimeWorldAudioBackendSessionContract.cpp")
    set_property(TARGET ${_audio_session_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_audio_session_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_audio_session_test} PRIVATE cxx_std_20)
    target_compile_options(${_audio_session_test} PRIVATE "$<TARGET_GENEX_EVAL:${_audio_session_owner},$<TARGET_PROPERTY:${_audio_session_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_audio_session_test} PRIVATE "$<TARGET_GENEX_EVAL:${_audio_session_owner},$<TARGET_PROPERTY:${_audio_session_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_audio_session_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_audio_session_crt}")
    endif()
    target_link_libraries(${_audio_session_test} PRIVATE concord_world_audio_backend_session_${_audio_session_profile}_leaf)
endforeach()
