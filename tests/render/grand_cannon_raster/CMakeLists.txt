cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ConcordGrandCannonRaster LANGUAGES CXX)
    set(S35_STANDALONE ON)
endif()
get_filename_component(S35_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${S35_SOURCE_ROOT}/cmake/CompilerContract.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
if(S35_STANDALONE)
    add_library(concord_grand_cannon_raster STATIC "${S35_SOURCE_ROOT}/src/render/voxel/GrandCannonRasterTask.cpp")
    add_library(concord_bounded_task_executor STATIC "${S35_SOURCE_ROOT}/src/core/tasks/BoundedTaskExecutor.cpp")
    target_include_directories(concord_bounded_task_executor PRIVATE "${S35_SOURCE_ROOT}/src")
    set_property(TARGET concord_bounded_task_executor PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
else()
    include("${S35_SOURCE_ROOT}/cmake/ProfiledLeaf.cmake")
    concord_add_profiled_leaf(concord_grand_cannon_raster "${S35_SOURCE_ROOT}/src/render/voxel/GrandCannonRasterTask.cpp" concord_runtime Runtime)
endif()
target_include_directories(concord_grand_cannon_raster PRIVATE "${S35_SOURCE_ROOT}/src")
add_executable(concord_grand_cannon_raster_contract GrandCannonRasterContract.cpp)
target_include_directories(concord_grand_cannon_raster_contract PRIVATE "${S35_SOURCE_ROOT}/src")
target_link_libraries(concord_grand_cannon_raster_contract PRIVATE concord_grand_cannon_raster concord_bounded_task_executor)
find_package(Threads REQUIRED)
target_link_libraries(concord_grand_cannon_raster_contract PRIVATE Threads::Threads)
foreach(s35_target IN ITEMS concord_grand_cannon_raster concord_grand_cannon_raster_contract)
    target_compile_features(${s35_target} PRIVATE cxx_std_20)
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${s35_target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    endif()
    if(NOT S35_STANDALONE AND s35_target STREQUAL "concord_grand_cannon_raster_contract")
        target_compile_options(${s35_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime,$<TARGET_PROPERTY:concord_runtime,COMPILE_OPTIONS>>")
        target_compile_definitions(${s35_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime,$<TARGET_PROPERTY:concord_runtime,COMPILE_DEFINITIONS>>")
    endif()
    if(CONCORD_MSVC_FRONTEND)
        concord_target_compile_options(${s35_target} PRIVATE /utf-8 /W4 /WX /permissive-)
    endif()
endforeach()
if(S35_STANDALONE)
    enable_testing()
    add_test(NAME grand_cannon_raster_contract_test COMMAND concord_grand_cannon_raster_contract)
    set_tests_properties(grand_cannon_raster_contract_test PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE LABELS "no-assets;bounded-tasks;render")
endif()
