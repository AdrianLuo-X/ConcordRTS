if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
add_executable(concord_software_frame_contract "${PROJECT_SOURCE_DIR}/tests/rendering/SoftwareFrameContract.cpp")

    set_property(TARGET concord_software_frame_contract PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)

    concord_target_compile_options(concord_software_frame_contract PRIVATE /W4 /WX)
