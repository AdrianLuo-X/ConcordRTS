if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_vision_profile IN ITEMS runtime contract)
    if(_vision_profile STREQUAL "runtime")
        set(_vision_owner concord_runtime)
        set(_vision_crt MultiThreadedDLL)
    else()
        set(_vision_owner concord_radar_cell_values_value_contract)
        set(_vision_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_vision_test concord_vision_phase_${_vision_profile}_contract)
    add_executable(${_vision_test} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeVisionPhaseContract.cpp")
    set_property(TARGET ${_vision_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_vision_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_vision_test} PRIVATE cxx_std_20)
    target_compile_options(${_vision_test} PRIVATE "$<TARGET_GENEX_EVAL:${_vision_owner},$<TARGET_PROPERTY:${_vision_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_vision_test} PRIVATE "$<TARGET_GENEX_EVAL:${_vision_owner},$<TARGET_PROPERTY:${_vision_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_vision_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_vision_crt}")
    endif()
    target_link_libraries(${_vision_test} PRIVATE concord_vision_phase_${_vision_profile}_leaf)
endforeach()
