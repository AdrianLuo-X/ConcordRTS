if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_super_profile IN ITEMS runtime contract)
    if(_super_profile STREQUAL "runtime")
        set(_super_owner concord_runtime)
        set(_super_crt MultiThreadedDLL)
    else()
        set(_super_owner concord_superweapon_smoke)
        set(_super_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_super_target concord_superweapon_core_${_super_profile}_contract)
    add_executable(${_super_target} "${PROJECT_SOURCE_DIR}/tests/infrastructure/SuperWeaponCoreContract.cpp")
    set_property(TARGET ${_super_target} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_super_target} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_super_target} PRIVATE cxx_std_20)
    target_compile_options(${_super_target} PRIVATE "$<TARGET_GENEX_EVAL:${_super_owner},$<TARGET_PROPERTY:${_super_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_super_target} PRIVATE "$<TARGET_GENEX_EVAL:${_super_owner},$<TARGET_PROPERTY:${_super_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_super_target} PROPERTY MSVC_RUNTIME_LIBRARY "${_super_crt}")
    endif()
    target_link_libraries(${_super_target} PRIVATE concord_superweapon_core_${_super_profile}_leaf)
endforeach()
