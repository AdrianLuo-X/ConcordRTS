if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_persistence_profile IN ITEMS runtime contract)
    if(_persistence_profile STREQUAL "runtime")
        set(_persistence_owner concord_runtime)
        set(_persistence_crt MultiThreadedDLL)
    else()
        set(_persistence_owner concord_radar_cell_values_value_contract)
        set(_persistence_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_persistence_test concord_trigger_persistence_${_persistence_profile}_contract)
    add_executable(${_persistence_test} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeTriggerPersistenceContract.cpp")
    set_property(TARGET ${_persistence_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_persistence_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_persistence_test} PRIVATE cxx_std_20)
    target_compile_options(${_persistence_test} PRIVATE "$<TARGET_GENEX_EVAL:${_persistence_owner},$<TARGET_PROPERTY:${_persistence_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_persistence_test} PRIVATE "$<TARGET_GENEX_EVAL:${_persistence_owner},$<TARGET_PROPERTY:${_persistence_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_persistence_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_persistence_crt}")
    endif()
    target_link_libraries(${_persistence_test} PRIVATE concord_trigger_persistence_${_persistence_profile}_leaf)
endforeach()
