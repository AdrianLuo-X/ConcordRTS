if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()

foreach(_tactics_profile IN ITEMS runtime contract)
    if(_tactics_profile STREQUAL "runtime")
        set(_tactics_owner concord_runtime)
        set(_tactics_crt MultiThreadedDLL)
    else()
        set(_tactics_owner concord_computer_mining_codec_contract)
        set(_tactics_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_tactics_target concord_computer_tactics_codec_${_tactics_profile})
    add_executable(${_tactics_target} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeComputerTacticsCodecContract.cpp")
    set_property(TARGET ${_tactics_target} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_tactics_target} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_tactics_target} PRIVATE cxx_std_20)
    target_compile_options(${_tactics_target} PRIVATE
        "$<TARGET_GENEX_EVAL:${_tactics_owner},$<TARGET_PROPERTY:${_tactics_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_tactics_target} PRIVATE
        "$<TARGET_GENEX_EVAL:${_tactics_owner},$<TARGET_PROPERTY:${_tactics_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_tactics_target} PROPERTY MSVC_RUNTIME_LIBRARY "${_tactics_crt}")
    endif()
    target_link_libraries(${_tactics_target} PRIVATE concord_computer_tactics_${_tactics_profile}_leaf)
endforeach()
