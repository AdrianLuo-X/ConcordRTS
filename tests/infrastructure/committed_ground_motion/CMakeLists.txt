if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_ground_profile IN ITEMS runtime contract)
    if(_ground_profile STREQUAL "runtime")
        set(_ground_owner concord_runtime)
        set(_ground_crt MultiThreadedDLL)
    else()
        set(_ground_owner concord_radar_cell_values_value_contract)
        set(_ground_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_ground_test concord_committed_ground_${_ground_profile}_contract)
    add_executable(${_ground_test} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeCommittedGroundMotionContract.cpp")
    set_property(TARGET ${_ground_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_ground_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_ground_test} PRIVATE cxx_std_20)
    target_compile_options(${_ground_test} PRIVATE "$<TARGET_GENEX_EVAL:${_ground_owner},$<TARGET_PROPERTY:${_ground_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_ground_test} PRIVATE "$<TARGET_GENEX_EVAL:${_ground_owner},$<TARGET_PROPERTY:${_ground_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_ground_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_ground_crt}")
    endif()
    target_link_libraries(${_ground_test} PRIVATE concord_committed_ground_${_ground_profile}_leaf)
endforeach()
