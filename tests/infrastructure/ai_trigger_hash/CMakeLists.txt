if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_hash_profile IN ITEMS runtime contract)
    if(_hash_profile STREQUAL "runtime")
        set(_hash_owner concord_runtime)
        set(_hash_crt MultiThreadedDLL)
    else()
        set(_hash_owner concord_computer_mining_codec_contract)
        set(_hash_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_hash_target concord_computer_ai_trigger_hash_${_hash_profile}_contract)
    add_executable(${_hash_target} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeComputerAiTriggerHashContract.cpp")
    set_property(TARGET ${_hash_target} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_hash_target} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_hash_target} PRIVATE cxx_std_20)
    target_compile_options(${_hash_target} PRIVATE "$<TARGET_GENEX_EVAL:${_hash_owner},$<TARGET_PROPERTY:${_hash_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_hash_target} PRIVATE "$<TARGET_GENEX_EVAL:${_hash_owner},$<TARGET_PROPERTY:${_hash_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_hash_target} PROPERTY MSVC_RUNTIME_LIBRARY "${_hash_crt}")
    endif()
    target_link_libraries(${_hash_target} PRIVATE concord_computer_ai_trigger_hash_${_hash_profile}_leaf)
endforeach()
