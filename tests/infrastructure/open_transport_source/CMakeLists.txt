if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_transport_profile IN ITEMS runtime contract)
    if(_transport_profile STREQUAL "runtime")
        set(_transport_owner concord_runtime)
        set(_transport_crt MultiThreadedDLL)
    else()
        set(_transport_owner concord_radar_cell_values_value_contract)
        set(_transport_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_transport_test concord_open_transport_source_${_transport_profile}_contract)
    add_executable(${_transport_test} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeOpenTransportSourceContract.cpp")
    set_property(TARGET ${_transport_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_transport_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_transport_test} PRIVATE cxx_std_20)
    target_compile_options(${_transport_test} PRIVATE "$<TARGET_GENEX_EVAL:${_transport_owner},$<TARGET_PROPERTY:${_transport_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_transport_test} PRIVATE "$<TARGET_GENEX_EVAL:${_transport_owner},$<TARGET_PROPERTY:${_transport_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_transport_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_transport_crt}")
    endif()
    target_link_libraries(${_transport_test} PRIVATE concord_open_transport_source_${_transport_profile}_leaf)
endforeach()
