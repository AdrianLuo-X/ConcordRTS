if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_identity_profile IN ITEMS runtime contract)
    if(_identity_profile STREQUAL "runtime")
        set(_identity_owner concord_runtime)
        set(_identity_crt MultiThreadedDLL)
    else()
        set(_identity_owner concord_radar_cell_values_value_contract)
        set(_identity_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_identity_test concord_save_identity_${_identity_profile}_contract)
    add_executable(${_identity_test} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeSaveIdentityContract.cpp")
    set_property(TARGET ${_identity_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_identity_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_identity_test} PRIVATE cxx_std_20)
    target_compile_options(${_identity_test} PRIVATE "$<TARGET_GENEX_EVAL:${_identity_owner},$<TARGET_PROPERTY:${_identity_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_identity_test} PRIVATE "$<TARGET_GENEX_EVAL:${_identity_owner},$<TARGET_PROPERTY:${_identity_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_identity_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_identity_crt}")
    endif()
    target_link_libraries(${_identity_test} PRIVATE concord_save_identity_${_identity_profile}_leaf)
endforeach()
