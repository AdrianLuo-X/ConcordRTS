if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_trigger_profile IN ITEMS runtime contract)
    if(_trigger_profile STREQUAL "runtime")
        set(_trigger_owner concord_runtime)
        set(_trigger_crt MultiThreadedDLL)
        set(_trigger_world concord_world_core_runtime_leaf)
    else()
        set(_trigger_owner concord_radar_cell_values_value_contract)
        set(_trigger_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
        set(_trigger_world concord_world_core_hash_contract_leaf)
    endif()
    set(_trigger_test concord_trigger_scheduler_${_trigger_profile}_contract)
    add_executable(${_trigger_test} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeTriggerSchedulerContract.cpp")
    set_property(TARGET ${_trigger_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_trigger_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_trigger_test} PRIVATE cxx_std_20)
    target_compile_options(${_trigger_test} PRIVATE "$<TARGET_GENEX_EVAL:${_trigger_owner},$<TARGET_PROPERTY:${_trigger_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_trigger_test} PRIVATE "$<TARGET_GENEX_EVAL:${_trigger_owner},$<TARGET_PROPERTY:${_trigger_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_trigger_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_trigger_crt}")
    endif()
    target_link_libraries(${_trigger_test} PRIVATE concord_trigger_scheduler_${_trigger_profile}_leaf ${_trigger_world})
endforeach()
