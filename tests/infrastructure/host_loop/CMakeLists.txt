cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ConcordHostLoop LANGUAGES CXX)
    set(W008_LOOP_STANDALONE ON)
endif()
get_filename_component(W008_LOOP_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${W008_LOOP_SOURCE_ROOT}/cmake/CompilerContract.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
add_executable(concord_host_loop_contract HostLoopContract.cpp)
set(w008_targets concord_host_loop_contract)
if(WIN32 AND NOT CONCORD_PORTABLE_CORE)
    add_executable(concord_host_loop_win32_contract HostLoopWin32Contract.cpp)
    if(TARGET concord_window_host_win32_contract_leaf)
        target_link_libraries(concord_host_loop_win32_contract PRIVATE concord_window_host_win32_contract_leaf)
    else()
        target_sources(concord_host_loop_win32_contract PRIVATE "${W008_LOOP_SOURCE_ROOT}/src/runtime/window/win32/RuntimeWindowHostWin32.cpp")
    endif()
    target_link_libraries(concord_host_loop_win32_contract PRIVATE user32)
    list(APPEND w008_targets concord_host_loop_win32_contract)
endif()
foreach(w008_target IN LISTS w008_targets)
    target_include_directories(${w008_target} PRIVATE "${W008_LOOP_SOURCE_ROOT}/src")
    target_compile_features(${w008_target} PRIVATE cxx_std_20)
    if(TARGET concord_runtime_match_score_contract)
        target_compile_options(${w008_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_OPTIONS>>")
        target_compile_definitions(${w008_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_DEFINITIONS>>")
    endif()
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${w008_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    if(CONCORD_MSVC_FRONTEND)
        concord_target_compile_options(${w008_target} PRIVATE /utf-8 /W4 /WX /permissive-)
    else()
        target_compile_options(${w008_target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endforeach()
if(W008_LOOP_STANDALONE)
    enable_testing()
endif()
add_test(NAME host_loop_contract_test COMMAND concord_host_loop_contract)
set_tests_properties(host_loop_contract_test PROPERTIES TIMEOUT 30 LABELS "no-assets;host-loop;contract")
if(TARGET concord_host_loop_win32_contract)
    add_test(NAME host_loop_win32_contract_test COMMAND concord_host_loop_win32_contract)
    set_tests_properties(host_loop_win32_contract_test PROPERTIES TIMEOUT 30 LABELS "windows;no-assets;host-loop;native-reference")
endif()
