if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_load_profile IN ITEMS runtime contract)
    if(_load_profile STREQUAL "runtime")
        set(_load_owner concord_runtime)
        set(_load_crt MultiThreadedDLL)
    else()
        set(_load_owner concord_superweapon_smoke)
        set(_load_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_load_target concord_load_transaction_${_load_profile}_contract)
    add_executable(${_load_target} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeLoadTransactionContract.cpp")
    set_property(TARGET ${_load_target} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_load_target} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_load_target} PRIVATE cxx_std_20)
    target_compile_options(${_load_target} PRIVATE "$<TARGET_GENEX_EVAL:${_load_owner},$<TARGET_PROPERTY:${_load_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_load_target} PRIVATE "$<TARGET_GENEX_EVAL:${_load_owner},$<TARGET_PROPERTY:${_load_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_load_target} PROPERTY MSVC_RUNTIME_LIBRARY "${_load_crt}")
    endif()
    target_link_libraries(${_load_target} PRIVATE concord_load_transaction_${_load_profile}_leaf)
endforeach()
