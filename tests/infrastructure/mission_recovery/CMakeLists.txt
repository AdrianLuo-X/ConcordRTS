if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_mission_recovery_profile IN ITEMS runtime contract)
    if(_mission_recovery_profile STREQUAL "runtime")
        set(_mission_recovery_owner concord_runtime)
        set(_mission_recovery_crt MultiThreadedDLL)
    else()
        set(_mission_recovery_owner concord_radar_cell_values_value_contract)
        set(_mission_recovery_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_mission_recovery_test concord_mission_recovery_${_mission_recovery_profile}_contract)
    add_executable(${_mission_recovery_test} "${PROJECT_SOURCE_DIR}/tests/infrastructure/RuntimeTeamMissionDelimitedRecoveryContract.cpp")
    set_property(TARGET ${_mission_recovery_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_mission_recovery_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_mission_recovery_test} PRIVATE cxx_std_20)
    target_compile_options(${_mission_recovery_test} PRIVATE "$<TARGET_GENEX_EVAL:${_mission_recovery_owner},$<TARGET_PROPERTY:${_mission_recovery_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_mission_recovery_test} PRIVATE "$<TARGET_GENEX_EVAL:${_mission_recovery_owner},$<TARGET_PROPERTY:${_mission_recovery_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_mission_recovery_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_mission_recovery_crt}")
    endif()
    target_link_libraries(${_mission_recovery_test} PRIVATE concord_trigger_persistence_${_mission_recovery_profile}_leaf concord_save_document_${_mission_recovery_profile}_leaf)
endforeach()
