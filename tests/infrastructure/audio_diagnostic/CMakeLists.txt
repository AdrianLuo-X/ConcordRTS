include("${CMAKE_SOURCE_DIR}/cmake/CompilerContract.cmake")
if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
add_executable(concord_audio_diagnostic_value_contract AudioDiagnosticValueContract.cpp)
set(w004_targets concord_audio_diagnostic_value_contract)
if(WIN32 AND NOT CONCORD_PORTABLE_CORE)
    add_library(concord_audio_diagnostic_native_leaf STATIC "${CMAKE_SOURCE_DIR}/src/audio/backend/RuntimeVoiceAudio.cpp")
    target_link_libraries(concord_audio_diagnostic_native_leaf PRIVATE xaudio2 ole32 xapobase)
    add_executable(concord_audio_diagnostic_native_contract AudioDiagnosticNativeContract.cpp)
    add_executable(concord_audio_diagnostic_original_spatial "${CMAKE_SOURCE_DIR}/tests/audio/RuntimeSpatialAudioSmoke.cpp")
    target_link_libraries(concord_audio_diagnostic_native_contract PRIVATE concord_audio_diagnostic_native_leaf)
    target_link_libraries(concord_audio_diagnostic_original_spatial PRIVATE concord_audio_diagnostic_native_leaf)
    list(APPEND w004_targets concord_audio_diagnostic_native_leaf concord_audio_diagnostic_native_contract concord_audio_diagnostic_original_spatial)
endif()
foreach(w004_target IN LISTS w004_targets)
    target_include_directories(${w004_target} PRIVATE "${CMAKE_SOURCE_DIR}/src")
    target_compile_features(${w004_target} PRIVATE cxx_std_20)
    if(w004_target STREQUAL "concord_audio_diagnostic_native_leaf")
        # Compile the unchanged native production source with its exact producer profile.
        target_compile_options(${w004_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime,$<TARGET_PROPERTY:concord_runtime,COMPILE_OPTIONS>>")
        target_compile_definitions(${w004_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime,$<TARGET_PROPERTY:concord_runtime,COMPILE_DEFINITIONS>>")
        concord_runtime_crt_profile(${w004_target})
        continue()
    endif()
    if(TARGET concord_runtime_match_score_contract)
        target_compile_options(${w004_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_OPTIONS>>")
        target_compile_definitions(${w004_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_DEFINITIONS>>")
    endif()
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${w004_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    if(CONCORD_MSVC_FRONTEND)
        concord_target_compile_options(${w004_target} PRIVATE /utf-8 /W4 /WX /permissive-)
    else()
        target_compile_options(${w004_target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    if(NOT w004_target STREQUAL "concord_audio_diagnostic_value_contract")
        concord_runtime_abi_definition(${w004_target})
        concord_runtime_crt_profile(${w004_target})
    endif()
endforeach()
