cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ConcordTimerResolution LANGUAGES CXX)
    set(W008_TIMER_STANDALONE ON)
endif()
get_filename_component(W008_TIMER_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${W008_TIMER_SOURCE_ROOT}/cmake/CompilerContract.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
add_executable(concord_timer_resolution_contract TimerResolutionContract.cpp)
set(w008_targets concord_timer_resolution_contract)
if(WIN32 AND NOT CONCORD_PORTABLE_CORE)
    add_executable(concord_timer_resolution_win32_contract TimerResolutionWin32Contract.cpp)
    target_link_libraries(concord_timer_resolution_win32_contract PRIVATE winmm)
    list(APPEND w008_targets concord_timer_resolution_win32_contract)
endif()
foreach(w008_target IN LISTS w008_targets)
    target_include_directories(${w008_target} PRIVATE "${W008_TIMER_SOURCE_ROOT}/src")
    target_compile_features(${w008_target} PRIVATE cxx_std_20)
    if(TARGET concord_runtime_match_score_contract)
        target_compile_options(${w008_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_OPTIONS>>")
        target_compile_definitions(${w008_target} PRIVATE "$<TARGET_GENEX_EVAL:concord_runtime_match_score_contract,$<TARGET_PROPERTY:concord_runtime_match_score_contract,COMPILE_DEFINITIONS>>")
    endif()
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${w008_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    if(CONCORD_MSVC_FRONTEND)
        concord_target_compile_options(${w008_target} PRIVATE /utf-8 /W4 /WX /permissive-)
    else()
        target_compile_options(${w008_target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endforeach()
if(W008_TIMER_STANDALONE)
    enable_testing()
endif()
add_test(NAME timer_resolution_contract_test COMMAND concord_timer_resolution_contract)
set_tests_properties(timer_resolution_contract_test PROPERTIES TIMEOUT 30 LABELS "no-assets;timer-resolution;contract")
if(TARGET concord_timer_resolution_win32_contract)
    add_test(NAME timer_resolution_win32_contract_test COMMAND concord_timer_resolution_win32_contract)
    set_tests_properties(timer_resolution_win32_contract_test PROPERTIES TIMEOUT 30 LABELS "windows;no-assets;timer-resolution;native-reference")
endif()

if(WIN32 AND TARGET concord_runtime)
    add_test(NAME runtime_timer_video_scope_contract_test COMMAND concord_runtime
        --command-smoke-test --resource-root "${W008_TIMER_SOURCE_ROOT}/RedAlert2"
        --local-data-root "${W008_TIMER_SOURCE_ROOT}/local_data"
        --map "${W008_TIMER_SOURCE_ROOT}/local_data/resources/mix_exports/multimd/2peaks.map")
    set_tests_properties(runtime_timer_video_scope_contract_test PROPERTIES TIMEOUT 90
        ENVIRONMENT "CONCORD_COMMAND_SMOKE_TIMER_VIDEO_SCOPE_ONLY=1"
        LABELS "windows;requires-game-assets;video;timer-resolution;contract")
endif()
