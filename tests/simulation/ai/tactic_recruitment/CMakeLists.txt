if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_tactic_recruitment_profile IN ITEMS runtime contract)
    if(_tactic_recruitment_profile STREQUAL "runtime")
        set(_tactic_recruitment_owner concord_runtime)
        set(_tactic_recruitment_crt MultiThreadedDLL)
    else()
        set(_tactic_recruitment_owner concord_radar_cell_values_value_contract)
        set(_tactic_recruitment_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_tactic_recruitment_test concord_computer_tactic_recruitment_${_tactic_recruitment_profile}_contract)
    add_executable(${_tactic_recruitment_test}
        "${PROJECT_SOURCE_DIR}/tests/simulation/ai/tactic_recruitment/RuntimeComputerTacticRecruitmentContract.cpp")
    set_property(TARGET ${_tactic_recruitment_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_tactic_recruitment_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_tactic_recruitment_test} PRIVATE cxx_std_20)
    target_compile_options(${_tactic_recruitment_test} PRIVATE "$<TARGET_GENEX_EVAL:${_tactic_recruitment_owner},$<TARGET_PROPERTY:${_tactic_recruitment_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_tactic_recruitment_test} PRIVATE "$<TARGET_GENEX_EVAL:${_tactic_recruitment_owner},$<TARGET_PROPERTY:${_tactic_recruitment_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_tactic_recruitment_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_tactic_recruitment_crt}")
    endif()
    target_link_libraries(${_tactic_recruitment_test} PRIVATE concord_computer_tactic_recruitment_${_tactic_recruitment_profile}_leaf)
endforeach()
