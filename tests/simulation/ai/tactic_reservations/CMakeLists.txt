if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_tactic_reservations_profile IN ITEMS runtime contract)
    if(_tactic_reservations_profile STREQUAL "runtime")
        set(_tactic_reservations_owner concord_runtime)
        set(_tactic_reservations_crt MultiThreadedDLL)
    else()
        set(_tactic_reservations_owner concord_radar_cell_values_value_contract)
        set(_tactic_reservations_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_tactic_reservations_test concord_computer_tactic_reservations_${_tactic_reservations_profile}_contract)
    add_executable(${_tactic_reservations_test}
        "${PROJECT_SOURCE_DIR}/tests/simulation/ai/tactic_reservations/RuntimeComputerTacticReservationsContract.cpp"
        "${PROJECT_SOURCE_DIR}/tests/simulation/ai/tactic_reservations/RuntimeComputerTacticReservationsPrivateContract.cpp")
    set_property(TARGET ${_tactic_reservations_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_tactic_reservations_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_tactic_reservations_test} PRIVATE cxx_std_20)
    target_compile_options(${_tactic_reservations_test} PRIVATE "$<TARGET_GENEX_EVAL:${_tactic_reservations_owner},$<TARGET_PROPERTY:${_tactic_reservations_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_tactic_reservations_test} PRIVATE "$<TARGET_GENEX_EVAL:${_tactic_reservations_owner},$<TARGET_PROPERTY:${_tactic_reservations_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_tactic_reservations_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_tactic_reservations_crt}")
    endif()
    target_link_libraries(${_tactic_reservations_test} PRIVATE concord_computer_tactic_reservations_${_tactic_reservations_profile}_leaf)
endforeach()
