if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_base_plan_profile IN ITEMS runtime contract)
    if(_base_plan_profile STREQUAL "runtime")
        set(_base_plan_owner concord_runtime)
        set(_base_plan_crt MultiThreadedDLL)
    else()
        set(_base_plan_owner concord_radar_cell_values_value_contract)
        set(_base_plan_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_base_plan_test concord_computer_base_plan_${_base_plan_profile}_contract)
    add_executable(${_base_plan_test}
        "${PROJECT_SOURCE_DIR}/tests/simulation/ai/base_plan/RuntimeComputerBasePlanContract.cpp")
    set_property(TARGET ${_base_plan_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_base_plan_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_base_plan_test} PRIVATE cxx_std_20)
    target_compile_options(${_base_plan_test} PRIVATE "$<TARGET_GENEX_EVAL:${_base_plan_owner},$<TARGET_PROPERTY:${_base_plan_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_base_plan_test} PRIVATE "$<TARGET_GENEX_EVAL:${_base_plan_owner},$<TARGET_PROPERTY:${_base_plan_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_base_plan_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_base_plan_crt}")
    endif()
    target_link_libraries(${_base_plan_test} PRIVATE concord_computer_base_plan_${_base_plan_profile}_leaf)
endforeach()
