if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_animation_profile IN ITEMS runtime contract)
    if(_animation_profile STREQUAL "runtime")
        set(_animation_owner concord_runtime)
        set(_animation_crt MultiThreadedDLL)
    else()
        set(_animation_owner concord_radar_cell_values_value_contract)
        set(_animation_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    foreach(_animation_case IN ITEMS auxiliary_bytes animation_metadata)
        set(_animation_test concord_${_animation_case}_${_animation_profile}_contract)
        if(_animation_case STREQUAL "auxiliary_bytes")
            add_executable(${_animation_test} RuntimeAuxiliaryShpByteCacheContract.cpp)
            target_link_libraries(${_animation_test} PRIVATE concord_auxiliary_shp_bytes_${_animation_profile}_leaf)
        else()
            add_executable(${_animation_test} RuntimeAnimationMetadataContract.cpp
                "${PROJECT_SOURCE_DIR}/src/assets/codecs/ShpCodec.cpp")
            target_link_libraries(${_animation_test} PRIVATE concord_superweapon_animation_timing_${_animation_profile}_leaf)
        endif()
        set_property(TARGET ${_animation_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
        target_include_directories(${_animation_test} PRIVATE "${PROJECT_SOURCE_DIR}/src" "${PROJECT_SOURCE_DIR}/tests/assets")
        target_compile_features(${_animation_test} PRIVATE cxx_std_20)
        target_compile_options(${_animation_test} PRIVATE "$<TARGET_GENEX_EVAL:${_animation_owner},$<TARGET_PROPERTY:${_animation_owner},COMPILE_OPTIONS>>")
        target_compile_definitions(${_animation_test} PRIVATE "$<TARGET_GENEX_EVAL:${_animation_owner},$<TARGET_PROPERTY:${_animation_owner},COMPILE_DEFINITIONS>>")
        if(CONCORD_MSVC_ABI)
            set_property(TARGET ${_animation_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_animation_crt}")
        endif()
    endforeach()
endforeach()
