if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_nuclear_death_profile IN ITEMS runtime contract)
    if(_nuclear_death_profile STREQUAL "runtime")
        set(_nuclear_death_owner concord_runtime)
        set(_nuclear_death_crt MultiThreadedDLL)
    else()
        set(_nuclear_death_owner concord_radar_cell_values_value_contract)
        set(_nuclear_death_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_nuclear_death_test concord_nuclear_death_${_nuclear_death_profile}_contract)
    add_executable(${_nuclear_death_test}
        "${PROJECT_SOURCE_DIR}/tests/simulation/superweapons/nuclear_death/RuntimeNuclearDeathDetonationContract.cpp")
    set_property(TARGET ${_nuclear_death_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_nuclear_death_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_nuclear_death_test} PRIVATE cxx_std_20)
    target_compile_options(${_nuclear_death_test} PRIVATE "$<TARGET_GENEX_EVAL:${_nuclear_death_owner},$<TARGET_PROPERTY:${_nuclear_death_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_nuclear_death_test} PRIVATE "$<TARGET_GENEX_EVAL:${_nuclear_death_owner},$<TARGET_PROPERTY:${_nuclear_death_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_nuclear_death_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_nuclear_death_crt}")
    endif()
    target_link_libraries(${_nuclear_death_test} PRIVATE concord_nuclear_death_${_nuclear_death_profile}_leaf)
endforeach()
