if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_dom_control_profile IN ITEMS runtime contract)
    if(_dom_control_profile STREQUAL "runtime")
        set(_dom_control_owner concord_runtime)
        set(_dom_control_crt MultiThreadedDLL)
    else()
        set(_dom_control_owner concord_radar_cell_values_value_contract)
        set(_dom_control_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_dom_control_test concord_dom_control_${_dom_control_profile}_contract)
    add_executable(${_dom_control_test}
        "${PROJECT_SOURCE_DIR}/tests/simulation/psionic/dominator_control/RuntimeDominatorControlCompletionContract.cpp")
    set_property(TARGET ${_dom_control_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_dom_control_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_dom_control_test} PRIVATE cxx_std_20)
    target_compile_options(${_dom_control_test} PRIVATE "$<TARGET_GENEX_EVAL:${_dom_control_owner},$<TARGET_PROPERTY:${_dom_control_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_dom_control_test} PRIVATE "$<TARGET_GENEX_EVAL:${_dom_control_owner},$<TARGET_PROPERTY:${_dom_control_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_dom_control_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_dom_control_crt}")
    endif()
    target_link_libraries(${_dom_control_test} PRIVATE concord_dom_control_${_dom_control_profile}_leaf)
endforeach()
