if(CONCORD_MSVC_ABI)
    set(CMAKE_CXX_STANDARD_LIBRARIES "kernel32.lib")
endif()
foreach(_mind_overload_profile IN ITEMS runtime contract)
    if(_mind_overload_profile STREQUAL "runtime")
        set(_mind_overload_owner concord_runtime)
        set(_mind_overload_crt MultiThreadedDLL)
    else()
        set(_mind_overload_owner concord_radar_cell_values_value_contract)
        set(_mind_overload_crt "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    set(_mind_overload_test concord_mind_overload_${_mind_overload_profile}_contract)
    add_executable(${_mind_overload_test}
        "${PROJECT_SOURCE_DIR}/tests/simulation/psionic/overload_effects/RuntimeMindControlOverloadEffectsContract.cpp")
    set_property(TARGET ${_mind_overload_test} PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_include_directories(${_mind_overload_test} PRIVATE "${PROJECT_SOURCE_DIR}/src")
    target_compile_features(${_mind_overload_test} PRIVATE cxx_std_20)
    target_compile_options(${_mind_overload_test} PRIVATE "$<TARGET_GENEX_EVAL:${_mind_overload_owner},$<TARGET_PROPERTY:${_mind_overload_owner},COMPILE_OPTIONS>>")
    target_compile_definitions(${_mind_overload_test} PRIVATE "$<TARGET_GENEX_EVAL:${_mind_overload_owner},$<TARGET_PROPERTY:${_mind_overload_owner},COMPILE_DEFINITIONS>>")
    if(CONCORD_MSVC_ABI)
        set_property(TARGET ${_mind_overload_test} PROPERTY MSVC_RUNTIME_LIBRARY "${_mind_overload_crt}")
    endif()
    target_link_libraries(${_mind_overload_test} PRIVATE concord_mind_overload_${_mind_overload_profile}_leaf)
endforeach()
